% how an alerting system turns a sensor signal into alerts.
% Top row: the four stages. Bottom: one bar per window (made-up heights), a

% Slide-only (the report does not include it). Laid out on a fixed millimetre grid;
% no axis numbers and no measured values.

% --- The four stages -----------------------------------------------------------
\node[stage] (sig) at (17,40)
  {\textbf{1\enspace Sensor signal}\\[1pt]\scriptsize motor current, cut into short windows};

  {\textbf{2\enspace Scorer} (fixed)\\[1pt]\scriptsize one anomaly score per window; higher means more unusual};
\node[stage, draw=accent, very thick, fill=accent!8] (pol) at (89,40)
  {\textbf{3\enspace Policy}\\[1pt]\scriptsize alert if the score is above the threshold $\theta$};
\node[stage] (op) at (125,40)
  {\textbf{4\enspace Operator}\\[1pt]\scriptsize checks every alert it receives};
\draw[flow] (sig) -- (sco);
\draw[flow] (sco) -- (pol);
\draw[flow] (pol) -- (op);
